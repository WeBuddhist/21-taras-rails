སྒྲོལ་མ་ཉེ་ཤུ་རྩ་གཅིག་གི་རྣམ་བཤད་ཞེས་བྱ་བ་བཞུགས་སོ།།

གླེང་བརྗོད།

༄། །བོད་སྨར་ཁམས་གཙང་གཤིས་ཕ་ར་རིན་ཆེན་གླིང་གི་དགོན་ས་འདིར་རྗེ་བཙུན་སྒྲོལ་མའི་གནས་ཞེས་ཡོངས་སུ་གྲགས་ཤིང་། ཡུལ་དགོན་གཉིས་ཀྱིས་ལོ་རེ་བཞིན་སྒྲོལ་མ་འབུམ་ཐེར་རེ་གྲངས་གསོག་གནང་སྲོལ་ཡོད་པ་བཞིན་འབྲེལ་ཡོད་འགའ་ཤས་ཀྱིས་ཕྲན་ལ་སྒྲོལ་མའི་འགྲེལ་པ་གཅིག་འབྲི་ཐུབ་ན་ཡག་པོ་འདུག་ཅེས་བསྐུལ་མ་གནང་བར་བརྟེན། ཕྲན་གྱིས་སྒྲོལ་མ་ཉེ་ཤུ་རྩ་གཅིག་གི་་རྣམ་བཤད་འདི་འདྲ་ཞིག་བྲིས་པ་དེའི་རང་གཞན་ཀུན་ལ་རྗེ་བཙུན་འཕགས་མ་སྒྲོལ་མའི་ཐུགས་རྗེ་དང་བྱིན་རླབས་འཇུག་པའི་སྨོན་ལམ་དང་བཅས༎

ཕུན་ཚོགས་དགེ་ལེགས་བྱེ་བས་བསྐྲུན་པའི་སྐུ།། མཐའ་ཡས་འགྲོ་བའི་རེ་བ་སྐོང་བའི་གསུང་། མ་ལུས་ཤེས་བྱ་ཇི་བཞིན་གཟིགས་པའི་ཐུགས།། ཤྭཀྱའི་གཙོ་བོ་དེ་ལ་མགོས་ཕྱག་འཚལ༎ རབ་འབྱམས་རྒྱལ་བའི་མཁྱེན་བརྩེའི་འཕྲིན་ལས་རྣམས༎ རབ་མཛེས་ཡིད་འོང་ལྷ་མོའི་སྐུར་བསྟན་ནས།། རབ་མང་འགྲོ་ལ་ཚེ་དང་མཆོག་སྩོལ་མ༎ རབ་དཀར་ལྷ་མོ་ཨུཏྤལ་བསྣམས་ལ་བསྟོད༎

སྒྲོལ་མ་ཉེ་ཤུ་རྩ་གཅིག་གི་རྣམ་བཤད་ཞེས་བྱ་བ་བཞུགས་སོ༎

༄༅།། རབ་འབྱམས་རྒྱལ་བ་ཀུན་གྱི་འཕྲིན་ལས་ཀྱི།། གཅིག་བསྡུས་ཐུཌ་རྗེའི་གཏེར་ཆེན་རྗེ་བཙུན་མས༎ འདོད་ཀུན་སྟེར་མཛད་རྗེ་བཙུན་སྒྲོལ་མ་ཡི༎ ཞབས་ཀྱི་པདྨོར་གུས་པས་ཕྱག་འཚལ་ལོ།།

སྒྲོལ་མའི་ལོ་རྒྱུས་མདོར་བསྡུས།

སྔོན་གྱི་དུས་སུ་བཅོམ་ལྡན་འདས་དེ་བཞིན་གཤེགས་པ་རྔ་སྒྲ་ཞེས་བྱ་བ་བྱུང་སྟེ། དེའི་བསྟན་པ་ལ་མཆོག་ཏུ་དད་པའི་རྒྱལ་པོའི་བུ་མོ་ཡེ་ཤེས་ཟླ་བ་ཞེས་བྱ་བ་དེས། སངས་རྒྱས་རྔ་སྒྲའི་འཁོར་བྱང་ཆུབ་སེམས་དཔའ་དང་ཉན་ཐོས་ཀྱི་དགེ་འདུན་སོགས་མང་པོ་ལ་ལོ་དུ་མར་མཆོད་པ་དང་བཀུར་སྟི་གྱ་ནོམ་བྱས་ཏེ་བྱང་ཆུབ་མཆོག་ཏུ་ཐུགས་དང་པོར་བསྐྱེད་པ་དང་། དེའི་ཚེ་བྱང་ཆུབ་སེམས་དཔའ་དང་ཉན་ཐོས་རྣམས་ཀྱིས་ཁྱེད་དགེ་བའི་རྩ་བ་འདི་ཉིད་ཀྱིས་ལུས་འདི་ཉིད་ལ་སྐྱེས་པར་ཕོར་གྱུར་ཏེ། བསྟན་པའི་བྱ་བ་བྱེད་པར་སྨོན་ལམ་བཏབ་ན་དེ་བཞིན་དུ་འགྱུར་བའི་ཕྱིར་དེ་ལྟར་རིགས་སོ་ཞེས་བསྐུལ་བ་དེའི་ལན་དུ། བུ་མོ་ཡེ་ཤེས་ཟླ་བས་སྐྱེས་པའི་རྟེན་ལ་བྱང་ཆུབ་འདོད་པ་ནི་མང་གི། བུད་མེད་ཀྱི་ལུས་རྟེན་གྱིས་འགྲོ་བའི་དོན་སྤྱོད་པ་ནི་འགའ་ཤས་ལས་མེད་པས། འཁོར་བ་མ་སྟོང་བར་དུ་བུད་མེད་ཀྱི་ལུས་རྟེན་གྱིས་འགྲོ་བ་སེམས་ཅན་རྣམས་ཀྱི་དོན་བྱའོ་ཞེས་དམ་བཅའ་མཛད་དོ།། དེ་ནས་རྒྱལ་པོའི་ཕོ་བྲང་དུ་བཞུགས་ཏེ། འདོད་པའི་ཡོན་ཏན་ལྔ་ནི། གཟུགས་སྒྲ་དྲི་རོ་རེག་བྱ་བཅས་ལ་ཐབས་མཁས་པས་ལོངས་སྤྱོད་བཞིན་དུ་ཏིང་ངེ་འཛིན་སྒོམ་པས་མི་སྐྱེ་བའི་ཆོས་ལ་བཟོད་པ་ཐོབ་ཅིང་། སེམས་ཅན་རྣམས་འཁོར་བ་ལས་སྒྲོལ་བ་ཞེས་བྱ་བའི་ཏིང་ངེ་འཛིན་བསྒྲུབས་པ་དེའི་ནུས་ པས་ཉིན་རེ་བཞིན་སེམས་ཅན་བརྒྱ་ཕྲག་མང་པོ་བདེ་བ་ལ་འགོད་པར་བྱེད་པས་སྔར་གྱི་མིང་དེ་འགྱུར་ནས་སྒྲོལ་མ་ཞེས་བྱ་བར་གྱུར་པ་སྟེ་དེ་བཞིན་གཤེགས་པ་རྔ་སྒྲས་ཀྱང་ཇི་སྲིད་བླ་ན་མེད་པའི་བྱང་ཆུབ་མངོན་དུ་བྱས་པ་དེ་སྲིད་དུ་མིང་ལྷ་མོ་སྒྲོལ་མའི་མཚན་འབའ་ཞིག་ཏུ་འགྱུར་བར་ལུང་བསྟན་ཏོ།།

སྒྲོལ་མའི་ཕན་ཡོན་ནི། རྒྱུད་ལས། སྲོད་དང་ཐོ་རངས་ལངས་པར་བྱས་ཏེ༎ དྲན་པས་མི་འཇིགས་ཐམས་ཅད་རབ་སྟེར༎ སྡིག་པ་ཐམས་ཅད་རབ་ཏུ་ཞི་བ༎ ངན་འགྲོ་ཐམས་ཅད་འཇོམས་པ་ཉིད་འཐོབ༎
