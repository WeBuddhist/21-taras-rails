[954] ཕྱག་འཚལ་ཉི་མ་ཟླ་བ་རྒྱས་པའི། །སྤྱན་གཉིས་པོ་ལ་འོད་རབ་
[955] གསལ་མ། ཧ་ར་གཉིས་བརྗོད་ཏུཏྟཱ་ར་ཡིས། །ཤིན་ཏུ་དྲག་པོའི་རིམས་

[959] རིམས་ནད་སེལ་བའི་སྨན་གྱི་བདུད་རྩིའི་ཟ་མ་ཏོག་འཛིན་པ གཡས་ཉི་

[963] ནད་ཡམས་རྣམས་སྲེག་པར་བྱེད་པ་དང་། གཡོན་ཟླ་བའི་སྤྱན་ལས་
[964] བདུད་རྩིའི་བསིལ་བའི་འོད་རབ་ཏུ་གསལ་བས་རིམས་དང་ཚ་བ་རྣམས་
[965] ཞི་བར་བྱེད་པ་མ་དང་། ཞལ་ལས་ཧཱ་ར་གཉིས་བརྗོད་ཏུཏྟཱ་ར་ཡི་ཞེས་ཨོཾ་
[966] ཏཱ་རེ་ཏུཏྟཱ་རེ་ན་མ་ཏྭ་ར་མ་ནོ་ཧཱ་ར་སྭཱ་ཧཱ། ཞེས་པའི་སྔགས་ཀྱིས་ཤིན་ཏུ་

[973] སྒྲོལ་མ་འོད་ཟེར་ཅན་མ་ལ ལྷ་མོ་དེ་ནི་སྐུ་མདོག་དཀར་མོ་གཡོན་

[975] པ་སྐུ་གསུང་ཐུགས་ཀྱི་གནས་གསུམ་སྟེ་སྤྱི་བོ། མགྲིན་པ། སྙིང་ག་

[977] བོན། ཨོཾ་དཀར་། ཨཱཿདམར་པོ། ཧཱུཾ་སྔོན་པོ་གསུམ་བཀོད་པའི་ཡི་གེ་
[978] དེ་ལས་འོད་ཟེར་འཕྲོས་པས་དད་པ་དང་ལྡན་པ་རྣམས་ཀྱིས་ལུས་ངག་
[979] ཡིད་གསུམ་ལ་འཚེ་བའི་ཕྱི་གཟུགས་ཅན་དང་གཟུགས་མེད་ཀྱི་བར་ཆད་
[980] དང་། ནང་ལུས་ཀྱི་ན་ཚ་འབྱུང་འཁྲུགས། གསང་བ་གཟུང་འཛིན་ཉོན་
[981] མོངས་རྣམས་ལྷ་མོ་ཁྱོད་ལ་ཞི་བའི་མཐུ་དང་ཡང་དག་པར་ལྡན་མ་དང་།

[995] དང་སྤེལ་བའི་བསྟོད་པ་འདི་དང་ཕྱག་འཚལ་བའི་ཚིགས་སུ་བཅད་པའི་

[997] སྒྲོལ་མ་ཉི་ཤུ་རྩ་གཅིག་གི་བསྟོད་པ་བསྟན་པའོ། །སྔོན་བྱོན་དམ་པའི་གསུང་གི་
[998] སྒྲོན་མེ་དང་། །འགྲོགས་པའི་རྣམ་དཔྱོད་འོད་ཀྱི་དྲྭ་བ་ཅན། །མོས་ལྡན་བློ་ཡི་མུན་པ་སེལ་བའི་
[999] ཕྱིར།།ལྷག་བསམ་གསེར་གྱི་ཞིང་རྟས་ལེགས་པར་དྲངས། །ཞེས་སྒྲོལ་མ་ཉེར་གཅིག་གི་བསྟོད་

[1013] དེ་ལ་འདིར་འཕགས་མའི་ཞབས་ལ་གསོལ་བ་བཏབས་པས་
[1014] འཇིགས་པ་རྣམས་ལས་བསྐྱབས་པའི་ལོ་རྒྱུས་ཡིད་ཆེས་ཆོག་པའི་གཏམ་

[1020] སྔོན་སྒྲོལ་མ་ཞེས་བྱ་བ་འཇིགས་པ་བཅུ་དྲུག་ལས་སྐྱོབ་པ་ཞིག་ཡོད་པར་
[1021] ཐོས་པ་དྲན་ནས། དེ་ལ་སྐྱབས་སུ་འགྲོའོ་བསམ་ནས་སྒྲོལ་མའི་མཚན་
[1022] ནས་བོས་ཤིང་གསོལ་བ་བཏབས་པས་དེ་མ་ཐག་སྒྲོལ་མ་མདུན་གྱི་ནམ་
[1023] མཁར་བྱོན་ནས་ཞབས་ཀྱི་འོག་ནས་རླུང་དྲག་པོ་གཡོས་པས་དམག་
[1024] རྣམས་སོ་སོར་འཐོར་བས་བདེ་བར་རང་ཁྱིམ་དུ་སླེབས་སྐད།
[1025] ༢༽ སེང་གེའི་འཇིགས་པ་ལས་བསྐྱབས་པ་སྔོན་རྒྱ་གར་གྱི་ས
[1026] ཆར་ཤིང་འཐུ་ཞིག་ཤིང་འཐུ་བར་ནགས་གསེབ་ལ་སོང་བས སེང་གེ་
[1027] མོ་བཀྲེས་པ་དང་འཕྲད ཟ་བར་བརྩམས་ནས་ཁ་ཡིས་ཁྱེར་ནས་སེང་
[1029] གེའི་ཚང་དང་ཉེ་བར་སླེབས་པ་ན། སྒྲོལ་མ་ལ་གསོལ་བ་རྩེ་གཅིག་
[1030] བཏབས་པས་མདུན་དུ་བུད་མེད་ཤིང་ལོའི་གོས་ཅན་ཞིག་བྱུང་ནས་སེང་
[1031] གེའི་ཁ་ནས་བླངས་ཏེ་གྲོང་ཁྱེར་གྱི་ཚོང་འདུས་སུ་བཞག་གོ།
[1032] ༣༽ གླང་པོའི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། བུ་མོ་ལོ་བཅུ་གཉིས་
[1033] ལོན་པ་ཞིག་མེ་ཏོག་འཐུ་བར་རང་ཉིད་གཅིག་པུར་སོང་བས། གླང་པོ་ཆེ་
[1034] གཏུམ་པོ་དང་འཕྲད་ནས་གླང་ཆེན་གྱི་སྣ་ཤན་གྱིས་བཅིངས། མཆེ་བར་
[1035] བསྣུན་ནས་གསོད་པའི་གྲབས་ཀྱི་ཚེ། སྒྲོལ་མའི་མཚན་དྲན་པར་བྱས་
[1036] ཤིང་གསོལ་བ་བཏབས་པས་དེ་མ་ཐག་གླང་ཆེན་གྱིས་བྱམས་པའི་ངང་
[1037] བཅིངས་པ་ལས་སློད་ནས་བདེ་བར་ཁྱིམ་ལ་བསྐྱལ་སྐད།
[1038] ༤༽ མེའི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། ཤ་འཁོན་ཆགས་པའི་ཁྱིམ་
[1039] ཚང་ཞིག་མཚན་མོར་དགྲ་ཡིས་མེ་བཏང་བ་ཁྱིམ་མི་རྣམས་བྲོས་པར་
[1040] བརྩམས་ཀྱང་མེས་བསྐོར་ནས་མ་ཐར་བས་ཅི་བྱ་བྲལ་སྐབས། རྗེ་བཙུན་
[1041] སྒྲོལ་མའི་མཚན་ནས་བོས་ཤིང་གསོལ་བ་བཏབས་པས། སྟེང་གི་ནམ་
[1042] མཁར་སྤྲིན་འཁྲིགས་ཤིང་ཆར་དྲག་ཏུ་བབས་པས་མེ་ཞི་བའོ།།
[1043] ༥༽ དུག་སྦྲུལ་གྱི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། སྔོན་གྲོང་ཁྱེར་
[1044] ཞིག་ན་སྨད་ཚོང་མ་ཞིག་ཡོད་པས། ནམ་ཕྱེད་ཙམ་ན་ཚོང་དཔོན་ལ་དུས་
[1045] བཏབས་ནས་ཚོང་དཔོན་གྱི་ཁྱིམ་དུ་འགྲོ་བའི་ཚེ་ལམ་དུ་སྐྱེས་པའི་ཤིང་
[1046] ཞིག་གི་ཡལ་ག་ལ་འཇུས་པ་ན། ཤིང་ལ་འཁྲིལ་བའི་སྦྲུལ་གདུག་པས་
[1047] ལུས་ལ་འཁྲིལ་ཞིང་ཟིན་པའི་ཚེ། འཇིགས་ཤིང་སྐྲག་པས་སྒྲོལ་མ་དྲན་
[1049] ནས་གསོལ་བ་བཏབས་པས་སྦྲུལ་གྱི་འཇིགས་པ་ལས་ཐར་ནས། སྦྲུལ་དེ་
[1050] དུག་མེད་པའི་སྦྲུལ་དུ་གྱུར་ནས་ཆུ་ལ་ཞུགས་པར་གྲགས།
[1051] ༦ ཆོམ་རྐུན་གྱི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། སྔོན་ཡུལ་ཞིག་ན་
[1052] ཚོང་དཔོན་ཤིན་ཏུ་ལོངས་སྤྱོད་ཆེན་པོ་ཡོད་པ་ཞིག །ཉིན་ཞིག་གཡོག་
[1053] འཁོར་དང་བཅས་ཚོང་ཟོང་མང་པོ་ཁྱེར་ཏེ་ལམ་དུ་འགྲོ་བའི་ཚེ། ཇག་པ་
[1054] ཆོམ་རྐུན་པ་མང་པོས་ཚོང་ཟོང་འཕྲོགས། གཡོག་འཁོར་རྣམས་བསད་
[1055] པའི་ཚེ་ཚོང་དཔོན་དེ་ཤིན་ཏུ་འཇིགས་ནས་མགོན་སྐྱབས་གཞན་མེད་
[1056] པས་སྒྲོལ་མ་ལ་གསོལ་བ་བཏབས་པ་ན། ཆོམ་རྐུན་རྣམས་འཇིགས་ཤིང་
[1057] བྲོས་པས་ཚོང་པ་རྣམས་ཆོམ་རྐུན་གྱི་འཇིགས་པ་ལས་གྲོལ།
[1058] ༧༽ བཙོན་རྭ་ལས་བསྐྱབས་པ་། རྐུན་པོའི་དཔོན་ཞིག་གིས་རྒྱལ་
[1059] པོའི་མཛོད་དུ་རྐུ་ལ་འགྲོ་བའི་ཚེ ཆང་ཞིམ་པོ་འཐུང་བས་ཆང་གི་ར་
[1060] བཟི་ནས་གཉིད་ལོག་ཏུ་སོང་། དེ་ནས་རྒྱལ་པོའི་མི་རྣམས་ཀྱིས་མཐོང་
[1061] ནས་རྐུ་མ་བཟུང་སྟེ་བཙོན་རྭར་བཅུག །ཆད་པ་སྣ་ཚོགས་པ་གཅད་ནས་
[1062] སྡུག་བསྔལ་བའི་ཚེ་སྒྲོལ་མ་ལ་གསོལ་བ་བཏབས་པས བྱ་ཁ་དོག་སྣ་
[1063] ཚོགས་པ་ཅན་ཞིག་བྱུང་ནས་བཅིངས་ཐག་རྣམས་དཀྲོལ། བཙོན་ཁང་གི་
[1064] སྒོ་ཡང་རང་བྱེ་སོང་ནས་རྐུན་མ་བཙོན་ལས་ཐར། རྨི་ལམ་དུ་བུད་མེད་
[1065] རྒྱན་སྣ་ཚོགས་པས་སྤྲས་པ་ཞིག་གིས་ཁྱེད་བཙོན་ལས་ཐར་བ་ང་ཡི་དྲིན་
[1066] ཡིན་པས། དྲིན་དྲན་ན་ཁྱེད་རང་འཁོར་དང་བཅས་པས་རྐུ་བྱེད་པའི་
[1067] ལས་སྤོངས་ཤིག་གསུངས།དེ་དག་ཚོང་སོགས་བྱ་བ་གཞན་ལས་རྐུ་བྱེད་
[1068] པའི་ལས་སྤངས་ནས་དགེ་བ་རྒྱ་ཆེན་པོ་བྱེད་པ་བྱུང་།
[1070] ༨༽ རྒྱ་མཚོའི་ཆུ་ཡི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། སྔོན་ཚོང་པ་
[1071] མང་པོ་རྒྱ་གར་ལྷོ་ཕྱོགས་ཀྱི་རྒྱ་མཚོའི་ཕྱི་རོལ་ཚོང་དུ་སོང་ནས་རྒྱ་
[1072] མཚོའི་ཕ་རོལ་ལ་ནོར་བུ་དང་ཙན་དན་སོགས་ཚོང་རྫས་རྣམས་བླངས་
[1073] ནས་མཚོ་ལ་ཞུགས་པའི་ཚེ། མཚོ་རླུང་ཆེན་པོ་དང་འགྲོགས་ནས་རྦ་
[1074] རླབས་འཁྲུགས་པ་གྲུ་དང་འཕྲད་ནས་གྲུ་འཐོར་ལ་ཉེ་ནས་ཅི་བྱ་བྲལ་
[1075] སྐབས་ཚོང་པ་རྣམས་ཀྱིས་ཚངས་པ་དང་། ཁྱབ་འཇུག །དབང་ཕྱུག་
[1076] སོགས་ལ་གསོལ་བ་བཏབ་ཀྱང་མ་ཕན་དེ་ནས་གྲུ་ཡི་ཐག་པ་ཆད། ཚོང་
[1077] རྫས་རྣམས་ཆུ་ལ་འཐོར་བའི་སྐབས། ཚོང་པའི་གྲལ་ལ་སངས་རྒྱས་ལ་
[1078] དད་པའི་དགེ་བསྙེན་གཅིག་ཡོད་པས། སྒྲོལ་མ་དྲན་ནས་ཡི་གེ་བཅུ་པ་
[1079] སྒྲོལ་མའི་སྔགས་བཟླས་ཤིང་གསོལ་བ་བཏབས་པས་དེ་མ་ཐག་རླུང་
[1080] རྣམས་ཞི། གྲུ་བསྐྱོད་པའི་མཐུན་ཕྱོགས་ཀྱི་རླུང་བྱུང་ནས་རྒྱ་མཚོའི་ཚུ་
[1081] རོལ་དུ་བདེ་བར་འབྱོར
[1082] ༩༽ ཤ་ཟའི་འཇིགས་པ་ལས་བསྐྱབས་པ་ནི། སྔོན་དགོན་པ་ཞིག་ན་
[1083] དགེ་འདུན་མང་པོ་ཡོད་པ་ལས་དུས་ཞིག་ན་དགོན་པ་ལས་མཚན་མོ་ཕྱི་
[1084] རོལ་ལ་སོང་བའི་དགེ་སློང་མང་པོ་འདས་པ་དང་། ཉིན་ཞིག་དགེ་ཚུལ་
[1085] ཞིག་ཕྱི་ལོག་འཆག་འཆག་ཏུ་སོང་བས ཤ་ཟ་འདྲེ་ངན་མདོག་ནག་ལ་
[1086] མཆེ་བ་གཙིགས་པ་གཅིག་བྱུང་བས་མགོ་བོ་ནས་བཟུང་བ་ན། དགེ་ཚུལ་
[1087] དེས་སྔོན་སྒྲོལ་མས་འཇིགས་པ་བརྒྱད་ལས་སྐྱོབ་པར་ཐོས་པས་སྒྲོལ་མ་
[1088] ཞེས་བོས་པ་དང་གསོལ་བ་བཏབས་པས་དེ་མ་ཐག་ལྷ་མོ་ནག་མོ་རལ་གྲི་
[1089] ཐོགས་པ་ཞིག་བྱུང་ནས་ཤ་ཟ་དེ་ལ་སྡིགས་པ་ན། ཤ་ཟ་དེ་སྐྲག་ནས་དགེ་
[1090] ཚུལ་ལ་བཟོད་གསོལ་བྱས་ནས་འཇིགས་པ་ལས་ཐར དེ་ཕྱིན་ཆད་
[1092] དགོན་པ་དེར་གནོད་པའི་རྒྱུན་ཆད་དོ།།
