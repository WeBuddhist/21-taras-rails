[159] ལས་ཉིད། །གཅིག་ཏུ་བསྡུས་པ་ཨརྱ་ཏཱ་རེར་འདུད། །གཡུ་མདོག་ལང་
[160] ཚོའི་མཛེས་སྡུག་ལྷུན་པོའི་སྐུ །མཚན་དཔེའི་དཔལ་གྱི་རི་མོ་འབུམ་

[166] རྟག་ཏུ་སྐྱབས་ཀྱི་ར་ལས་མི་འདའ་བར། །འཇིགས་བརྒྱད་བཅུ་དྲུག་

[169] དཔའ་བོ་འབུམ་གྱི་སྲིང་གཅིག་མ། །སྐལ་ལྡན་རིག་འཛིན་བདེ་སྟོང་
[170] བསྐྱེད་པའི་གྲོགས། །བྲང་ཀྱི་དཀྱིལ་ལ་མྱོས་བུམ་ཟླུམ་ཞིང་རྒྱས། །རིང་

[198] ཁེབས། །དྲི་ཞིམ་རླུང་གིས་ལྷབ་ལྷུབ་གཡོ་བའི་རྩེར། །རིགས་བདག་

[219] དང་མངོན་མཛད་རྟེན། །མཛེས་མའི་ཚུལ་ལ་འཆང་བའི་སྙིང་སྟོབས་ནི།།
[220] དུས་གསུམ་རྒྱལ་བ་ཀུན་གྱིས་འགྲན་མེད་ཐུགས། །དེ་ལྟར་བསྟོད་ཚིག་
[221] པད་དཀར་ཆུན་པོ་འདི། །འཕགས་མ་དགྱེས་པའི་མིག་ལ་མཛའ་བ་དང་།།
[222] བདག་ཀྱང་ཇི་སྲིད་བྱང་ཆུབ་མ་ཐོབ་བར། །འབྲལ་མེད་རྗེས་སུ་འཛིན་
[223] པའི་རྒྱུར་གྱུར་ཅིག །ཅེས་རྒྱལ་ཡུམ་འཕགས་མ་སྒྲོལ་མ་ཚིགས་སུ་བཅད་པ་བསྟོད་པའི་
[224] རྒྱལ་པོའི་གྲངས་དང་ལྡན་པས་མཆོད་པར་བསྟོད་པ་འདི་ནི་དལ་རྟེན་གྱི་ལུས་འདི་ལྷ་མཆོག་ཨརྱ་
[225] ཏཱ་རེས་བཀའ་དྲིན་གྱིས་འཚོ་བ་ཤྲིའི་མིང་མཐས་བརྒྱན་པའི་རྒྱལ་ཁམས་པ་དེས་ཤར་ཕྱོགས་པོ་
[227] ཊ་ལ་ཡི་གནས་ནས་བྲིས་པ་བཀྲ་ཤིས་དགེ་བའི་སྣང་བ་རྣམ་པར་རྒྱལ་བར་གྱུར་ཅིག།
[229] ༄༅། །རྗེ་བཙུན་སྒྲོལ་མའི་བསྟོད་པའི་འགྲེལ་པ་
[230] འཕགས་མའི་བྱིན་རླབས་འགུགས་པའི་ལྕགས་ཀྱུ་

[235] འཇིགས་དང་བར་ཆད་ཀུན་ལས་སྐྱོབ་བོ་ཞེས།།
[236] ཞལ་གྱིས་བཞེས་པའི་བརྩེ་ཆེན་གཡར་དམ་ཅན།།

[243] བར། སངས་རྒྱས་རྔ་སྒྲ་ཞེས་བྱ་བ་བྱུང་། དེའི་བསྟན་པ་ལ་དད་ཅིང་གུས་
[244] པའི་རྒྱལ་པོའི་བུ་མོ་ཡེ་ཤེས་ཟླ་བ་ཞེས་བྱ་བ་ཞིག་ཡོད།དེས་སངས་རྒྱས་
[245] དེ་དང་འཁོར་བྱང་ཆུབ་སེམས་དཔའི་དགེ་འདུན་གྲངས་མང་པོ་ལ་ལོ་
[246] བརྒྱ་སྟོང་མང་པོའི་བར་དུ་མཆོད་པ་དང་བཀུར་བསྟི་བྱས་མཐར་སངས་

[260] པོར་རྒྱལ་པོའི་ཕོ་བྲང་དུ་གནས་ནས་འདོད་པའི་ཡོན་ཏན་རྣམས་ལ་
[261] ཐབས་མཁས་ཀྱིས་ལོངས་སྤྱོད་བཞིན་ཏིང་ངེ་འཛིན་བསྒོམས་པས། ཆོས་
[262] ཐམས་ཅད་ལ་སྐྱེ་བ་མེད་པར་རྟོགས་པའི་མི་སྐྱེ་བའི་ཆོས་ལ་བཟོད་པ་

[272] ལམ་གྱི་སྐབས་དང་མཐར་སངས་རྒྱས་པའི་དུས་ཐམས་ཅད་ལ་མཚན་
[273] སྒྲོལ་མ་ཞེས་དང་། གཟུགས་བུད་མེད་ཀྱི་རྟེན་མཛད་པའོ། །སླར་ཡང་
[274] སངས་རྒྱས་དོན་ཡོད་གྲུབ་པའི་དྲུང་དུ སེམས་ཅན་རྣམས་ལ་ཉེ་བར་

[278] ནོ །འཇིག་རྟེན་གྱི་ཁམས་འདིར་སྣང་ཚུལ་ནི་འཕགས་པ་སྤྱན་རས་
